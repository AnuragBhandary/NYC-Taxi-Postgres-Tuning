-- JFK pickups by weekday and hour, whole year
SELECT WEEKDAY(day) + 1 AS dow, hour, SUM(trips) AS trips
                 FROM mv_hourly_zone WHERE pu_location_id = 132
                   AND day >= '2023-01-01' AND day < '2024-01-01'
                 GROUP BY 1, 2 ORDER BY 1, 2
